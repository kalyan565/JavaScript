Day 6 — Functions

Hands-on

Task 1 Rectangle area
  • Write function area(length, width) that returns the area of a rectangle.
  • Call it three times with different values and console.log each result.
  • Bonus: rewrite the same logic as an arrow function with implicit return.
function area(length, width) {
  return length * width;
}

console.log(area(10, 5));
console.log(area(7, 3));
console.log(area(15, 4));

const areaArrow = (length, width) => length * width;
console.log("arrow:", areaArrow(8, 2));

Reference: ../js/hands-on.js (loaded below).

Task 2 Greeting with default
  • Write function greet(name) that returns "Hello, <name>!" (use a template literal).
  • Use a default parameter: if no name is passed, default to "Guest".
  • Call it with "Priya", with "Aarav", and with no argument.
  • Bonus: try greet(null) — does it use the default? No: default parameters only apply when the argument is undefined (missing). null is an explicit value, so you get "Hello, null!".
function greet(name = "Guest") {
  return `Hello, ${name}!`;
}

console.log(greet("Priya"));
console.log(greet("Aarav"));
console.log(greet());
console.log(greet(null));

Reference: ../js/hands-on.js.

Task 3 Temperature converter
  • Write an arrow function cToF(celsius) with implicit return.
  • Formula: F = C * 9/5 + 32.
  • Test with: 0, 100, 37 (body temperature), 45 (Jaipur summer).
const cToF = (celsius) => (celsius * 9) / 5 + 32;

console.log(cToF(0));   // 32
console.log(cToF(100)); // 212
console.log(cToF(37));  // 98.6
console.log(cToF(45));  // 113

Reference: ../js/hands-on.js.

Bonus Pure vs impure
  • Write a pure function double(n) that returns n * 2.
  • Write an impure function addToTotal(n) that updates an outer let total = 0 and returns the new total.
  • Call each function three times. Notice how the impure output depends on earlier calls.
  • Add a comment: which is easier to reason about, and why?
function double(n) {
  return n * 2;
}
console.log("pure double:", double(2), double(5), double(10));

let total = 0;
function addToTotal(n) {
  total += n;
  return total;
}
console.log("impure addToTotal:", addToTotal(5), addToTotal(5), addToTotal(5));

/*
Pure double(n) is easier to reason about: same input → same output, no hidden state.
addToTotal mutates outer total, so output depends on call history.
*/
