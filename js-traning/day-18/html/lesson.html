Day 18 — Async: Callbacks & Promises

Lesson

Open DevTools → Console, or run node day-18/js/lesson.js (takes ~20s — timers). Source: ../js/lesson.js.

1 Sync vs async

Topic 1 — Sync vs async. Synchronous code runs top to bottom. Asynchronous scheduling (for example setTimeout) does not pause the rest of the script — callbacks run later on the event loop.

// SYNCHRONOUS — top to bottom, no surprises
console.log("1");
console.log("2");
console.log("3");
// Output: 1, 2, 3

// ASYNCHRONOUS — setTimeout schedules; JS doesn't wait
console.log("1");
setTimeout(() => console.log("2"), 0); // scheduled — runs LATER
console.log("3");
// Output: 1, 3, 2
//
// Even with delay 0! "2" goes through the event loop.
// JS keeps running synchronous code first; async callbacks wait their turn.

2 Callbacks

Topic 2 — Callbacks. Pass a function to be called when work finishes. Node-style APIs often use (err, data) as the callback arguments.

// Simulate fetching a user from a server
function fetchUser(id, callback) {
  console.log(`Fetching user ${id}...`);
  setTimeout(() => {
    const user = { id, name: "Priya" };
    callback(null, user); // Node-style: (err, data)
  }, 1000);
}

fetchUser(7, (err, user) => {
  if (err) {
    console.error("Failed:", err);
    return;
  }
  console.log("Got user:", user);
});
// Logs: Fetching user 7... then (after delay) Got user: { id: 7, name: "Priya" }

3 Callback hell

Topic 3 — Callback hell (pyramid of doom). Each step needs the previous result, so callbacks nest inward. That is why Promises (and later async/await) became central to modern JS.

function fetchUser(id, callback) {
  setTimeout(() => callback(null, { id, name: "Priya" }), 200);
}
function fetchOrders(userId, callback) {
  setTimeout(() => callback(null, [{ id: 101, userId }]), 200);
}
function fetchItems(orderId, callback) {
  setTimeout(() => callback(null, [{ sku: "A1", orderId }]), 200);
}

// The "pyramid of doom"
fetchUser(7, (err, user) => {
  if (err) { console.error(err); return; }
  fetchOrders(user.id, (err, orders) => {
    if (err) { console.error(err); return; }
    fetchItems(orders[0].id, (err, items) => {
      if (err) { console.error(err); return; }
      console.log(items); // finally got there!
    });
  });
});
// Problems: indentation, repeated errors, hard to extend,
// no clean "return" through async — only nest deeper.

4 Promise states

Topic 4 — Promise states. A promise starts pending, then becomes fulfilled (value) or rejected (reason). Either outcome is settled (terminal).

State Meaning
pendingWork in progress
fulfilledDone, has a result
rejectedFailed, has an error
settledEither fulfilled or rejected (terminal)
 
// Create a promise
const p = new Promise((resolve, reject) => {
  setTimeout(() => {
    const success = true;
    if (success) resolve("Done!"); // → fulfilled
    else reject(new Error("Oops")); // → rejected
  }, 1000);
});
// p is PENDING. After 1s, it settles.

// Real-world: wrapping setTimeout
function delay(ms) {
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}
delay(500).then(() => console.log("half a second passed"));

Follow along. If you log the promise immediately after new Promise, you still see a pending promise; .then runs after resolve(42).

const pFollow = new Promise((res) => {
  setTimeout(() => res(42), 100);
});
console.log(pFollow); // Promise { <pending> } — exact string varies by engine
pFollow.then((v) => console.log(v)); // 42 — after the timer

5 .then / .catch / .finally

Topic 5 — Chaining. Return a plain value from .then to pass it forward; return another promise and the chain waits for it. .catch handles any rejection above it. .finally always runs.

function fetchUser(id) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (id < 0) reject(new Error("Bad id"));
      else resolve({ id, name: "Priya" });
    }, 500);
  });
}

fetchUser(7)
  .then((user) => {
    console.log("got user:", user);
    return user.id;
  })
  .then((id) => {
    return fetchUser(id + 1); // returning a Promise — chain awaits it
  })
  .then((nextUser) => {
    console.log("next user:", nextUser);
  })
  .catch((err) => {
    console.error("any failure caught here:", err.message);
  })
  .finally(() => {
    console.log("done — runs whether success or fail");
  });

../js/lesson.js — Topic 5 (fetchUserPromise)

6 Promise combinators

Topic 6 — Combine many promises. Use the right combinator when you care about all results, first finish, or first success. In the browser, p1–p3 are often fetch(...) calls; the runnable lesson.js uses mock timers so the same ideas work in Node offline.

Combinator Resolves when Rejects when
Promise.allALL fulfillANY rejects (fail-fast)
Promise.allSettledALL settleNever
Promise.raceFIRST settlesIf the first to settle rejected
Promise.anyFIRST fulfillsALL reject (AggregateError)
 
const p1 = fetch("/user");
const p2 = fetch("/orders");
const p3 = fetch("/items");
// use delay(ms) from Topic 4, or: const delay = (ms) => new Promise((r) => setTimeout(r, ms));

// 1. all — wait for everything; fail if any fails
Promise.all([p1, p2, p3])
  .then(([user, orders, items]) => console.log("all done"))
  .catch((err) => console.error("at least one failed:", err));

// 2. allSettled — wait for everything; never throws as a whole
Promise.allSettled([p1, p2, p3]).then((results) => {
  results.forEach((r) => {
    if (r.status === "fulfilled") console.log("ok:", r.value);
    else console.log("fail:", r.reason);
  });
});

// 3. race — first to settle wins (success OR fail)
Promise.race([p1, delay(5000)])
  .then((winner) => console.log("first done:", winner));

// 4. any — first to FULFILL wins; all-reject = AggregateError
Promise.any([p1, p2, p3])
  .then((firstSuccess) => console.log("got one:", firstSuccess))
  .catch((err) => console.error("everyone failed"));

../js/lesson.js — Topic 6

7 Promise.resolve / Promise.reject

Topic 7 — Already-settled shortcuts. Wrap values or errors in a promise so callers always get the same async shape.

// assumes fetchUser(id) returns a Promise (see Topic 5 pattern)

// Already-fulfilled promise
const cached = Promise.resolve({ id: 1, name: "Priya" });
cached.then((u) => console.log(u));

// Already-rejected promise
const failed = Promise.reject(new Error("nope"));
failed.catch((e) => console.log("caught:", e.message));

// Common pattern: cache or fetch
function getUser(id, cache) {
  if (cache[id]) return Promise.resolve(cache[id]); // instant
  return fetchUser(id); // real async — must return a Promise
}
// Caller doesn't care which path — both return a Promise
getUser(7, {}).then(console.log);
