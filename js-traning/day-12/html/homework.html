Day 12 — Error Handling & Module Basics

Homework

Tasks 1–2: ../js/homework.js. Tasks 3–4: ../js/homework/calc.js + ../js/homework/app.js (ES module). Open the console (F12).

1 safeDivide(a, b) — divide by zero

Throw when b === 0. Callers wrap in try/catch and log err.message.

function safeDivide(a, b) {
  if (b === 0) throw new Error("Cannot divide by zero");
  return a / b;
}
try {
  console.log(safeDivide(12, 4));
} catch (err) {
  console.log(err.message);
}

Reference: ../js/homework.js

2 NotFoundError + getUserById(id)

Extend Error, set this.name = "NotFoundError". Throw if id is not 1, 2, or 3. In catch, branch on instanceof NotFoundError.

class NotFoundError extends Error {
  constructor(message) {
    super(message);
    this.name = "NotFoundError";
  }
}
function getUserById(id) {
  if (id !== 1 && id !== 2 && id !== 3) {
    throw new NotFoundError(`No user with id ${id}`);
  }
  return { id, name: `User ${id}` };
}

Reference: ../js/homework.js

3 Calculator module — named exports (homework/calc.js)

Export add, subtract, multiply, divide (throw on divide-by-zero).

export function add(a, b) { return a + b; }
export function subtract(a, b) { return a - b; }
export function multiply(a, b) { return a * b; }
export function divide(a, b) {
  if (b === 0) throw new Error("Cannot divide by zero");
  return a / b;
}

Reference: ../js/homework/calc.js

4 Default calculate + entry homework/app.js

Default export calculate(op, a, b) for + - * /. Import default + named bindings in the app entry (same pattern as app.js next to calc.js).

// homework/app.js
import calculate, { add, subtract, multiply, divide } from "./calc.js";
console.log(add(10, 4), calculate("+", 2, 3));

Reference: ../js/homework/calc.js, ../js/homework/app.js — loaded with <script type="module" src="../js/homework/app.js"></script>
