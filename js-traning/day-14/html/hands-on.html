Hands-on

Task 1 Build a Counter
  • Create a closure-based counter.
  • Build two independent counters.
  • Verify each counter maintains its own state. 
function makeCounter() {

  let count = 0;

  return function () {

    count++;

    return count;
  };
}

const counter1 =

  makeCounter();

const counter2 =

  makeCounter();

console.log(
  counter1()
);

console.log(
  counter1()
);

console.log(
  counter2()
);

console.log(
  counter2()
);
// Output:
//
// 1
// 2
// 1
// 2
//
// The count variable lives
// inside the closure created
// by makeCounter().
//
// Each counter gets its own
// private memory.

Task 2 Fix the var-in-Loop Bug
  • Observe the classic var loop issue.
  • Fix it using let.
  • Understand block scoping. 
// Problem

for (

  var i = 1;

  i <= 3;

  i++
) {

  setTimeout(() =>

    console.log(i),

    100
  );
}
// Output:
//
// 4
// 4
// 4
//
// var uses function scope.
//
// All callbacks share
// the same i variable.
// Fixed Version

for (

  let i = 1;

  i <= 3;

  i++
) {

  setTimeout(() =>

    console.log(i),

    100
  );
}
// Output:
//
// 1
// 2
// 3
//
// let creates a new block-scoped
// variable for every iteration.

Try the steps, then compare with ../js/hands-on.js.

Task 3 Private Bank Balance
  • Create a private balance using closures.
  • Add deposit and withdraw methods.
  • Prevent direct balance access. 
function createAccount(initial) {

  let balance = initial;

  return {

    deposit(amount) {

      balance += amount;
    },

    withdraw(amount) {

      balance -= amount;
    },

    getBalance() {

      return balance;
    }
  };
}

const account =

  createAccount(1000);

account.deposit(500);

account.withdraw(200);

console.log(
  account.getBalance()
);

console.log(
  account.balance
);
// Output:
//
// 1300
// undefined
//
// balance remains private
// inside the closure and
// cannot be accessed directly.

Try the steps, then compare with ../js/hands-on.js.

Bonus Build a Memoizer
  • Cache function results using closures.
  • Avoid repeated computations.
  • Understand persistent closure memory. 
function memoize(fn) {

  const cache = {};

  return function (n) {

    if (cache[n]) {

      return cache[n];
    }

    const result = fn(n);

    cache[n] = result;

    return result;
  };
}

function expensiveSquare(n) {

  console.log("computing...");

  return n * n;
}

const memoSquare =

  memoize(expensiveSquare);

console.log(
  memoSquare(5)
);

console.log(
  memoSquare(5)
);

console.log(
  memoSquare(10)
);

console.log(
  memoSquare(5)
);
// Output:
//
// computing...
// 25
// 25
// computing...
// 100
// 25
//
// The cache object lives
// inside the closure created
// by memoize().

Try the steps, then compare with ../js/hands-on.js.
